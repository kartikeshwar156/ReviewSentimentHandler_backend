class CommentSearchSystem:
    """
    Complete system that stores and retrieves original text.
    """
    
    def __init__(self):
        self.vectorizer = None
        self.comment_vectors = None
        self.original_texts = []  # Store original comments
        self.comment_ids = []     # Store IDs
    
    def train(self, comments, ids):
        """Train on comments and store everything."""
        # Store originals
        self.original_texts = comments
        self.comment_ids = ids
        
        # Train vectorizer and create vectors
        self.vectorizer = TfidfVectorizer(max_features=1000, ngram_range=(1, 2))
        self.comment_vectors = self.vectorizer.fit_transform(comments)
        
        print(f"✓ Trained on {len(comments)} comments")
    
    def search(self, query, top_k=5):
        """Search and return ORIGINAL TEXT of similar comments."""
        # Transform query
        query_vector = self.vectorizer.transform([query])
        
        # Calculate similarities
        similarities = cosine_similarity(query_vector, self.comment_vectors).flatten()
        
        # Get top-k indices
        top_indices = similarities.argsort()[-top_k:][::-1]
        
        # Return original text using indices
        results = []
        for idx in top_indices:
            results.append({
                'comment_id': self.comment_ids[idx],
                'similarity': float(similarities[idx]),
                'text': self.original_texts[idx]  # ← Original line!
            })
        
        return results
    
    def save(self):
        """Save vectorizer, vectors, and text mapping."""
        import pickle
        from scipy.sparse import save_npz
        
        # Save vectorizer
        with open('vectorizer.pkl', 'wb') as f:
            pickle.dump(self.vectorizer, f)
        
        # Save vectors
        save_npz('vectors.npz', self.comment_vectors)
        
        # Save text and IDs
        with open('comments_data.pkl', 'wb') as f:
            pickle.dump({
                'texts': self.original_texts,
                'ids': self.comment_ids
            }, f)
        
        print("✓ Saved all data to disk")
    
    def load(self):
        """Load everything from disk."""
        import pickle
        from scipy.sparse import load_npz
        
        # Load vectorizer
        with open('vectorizer.pkl', 'rb') as f:
            self.vectorizer = pickle.load(f)
        
        # Load vectors
        self.comment_vectors = load_npz('vectors.npz')
        
        # Load text and IDs
        with open('comments_data.pkl', 'rb') as f:
            data = pickle.load(f)
            self.original_texts = data['texts']
            self.comment_ids = data['ids']
        
        print("✓ Loaded all data from disk")
